cmake_minimum_required(VERSION 3.16)
project(emwm-compositor VERSION 0.2.7 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(X REQUIRED IMPORTED_TARGET x11 xcomposite xdamage xfixes xrender xext xrandr gl)
add_executable(emwm-compositor src/main.cpp src/renderer.cpp)
target_link_libraries(emwm-compositor PRIVATE PkgConfig::X)
target_compile_options(emwm-compositor PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
add_executable(test-effects tests/effects.cpp)
target_include_directories(test-effects PRIVATE src)
add_test(NAME effect-geometry COMMAND test-effects)
add_executable(test-render-effects tests/render-effects.cpp src/renderer.cpp)
target_include_directories(test-render-effects PRIVATE src)
target_link_libraries(test-render-effects PRIVATE PkgConfig::X)
target_compile_options(test-render-effects PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS emwm-compositor RUNTIME DESTINATION bin)
